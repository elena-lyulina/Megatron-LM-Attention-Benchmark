############ QWEN-STYLE HYBRID 1B CONFIG (param-matched to Llama-3.2-1B) #################
# Interleaves a GDN linear mixer with gated softmax attention on the LLaMA-3.2-1B backbone,
# replicating the Qwen3.5 / 3.8 hybrid structure (model_type "qwen3_5", full_attention_interval 4):
# 3 GDN layers then 1 gated-attention layer, repeated -> 12 GDN + 4 gated attn over 16 layers,
# gated-attn layers at indices 3, 7, 11, 15.
#
# Only the *hybridization pattern* is taken from Qwen. Each component is frozen to the pure
# baseline we already trained, so the hybrid stays comparable to both:
#   - GDN block   = gdn_1B_args_8heads_ffn5824.txt  (8 K/V heads, 192/384, paper ratios 0.75/1.5)
#   - gated block = gated_attn_llama_1B_args_ffn7488.txt  (Llama GQA 32/8 + --attention-output-gate, full RoPE scf=1)
# Deliberately NOT matched to Qwen: Qwen pins linear key heads at 16 / head_dim 128/128, uses
# fewer attn heads at head_dim 256 with partial rotary 0.25, MoE, and an MTP head -- all dropped.
#
# Target: Llama-3.2-1B = 1,235,814,400 params. This config: ~1,234,410,176 (-0.11%); run the
# counter to lock the exact total. Analytic exact-match FFN is ~6222; 6208 (97x64) is the nearest
# multiple of 64 under it, 6272 lands +0.40% over.
# args names taken from: megatron/core/transformer/transformer_config.py + megatron/training/arguments.py

### Embeddings / input-output:
--make-vocab-size-divisible-by 128      # sizes the model's input / output embedding matrix (vocab x hidden / hidden x vocab)
# Llama-3.2-1B config: "vocab_size": 128256 -- derived from the tokenizer
# Llama-3.2-1B config: "tie_word_embeddings": true -- embeddings tied by default; need --untie-embeddings-and-output-weights to disable

### Global / model-wide
# not changed, same as Llama 1B config
--hidden-size 2048                      # representing 1 token
--num-layers 16                         # how many blocks
--normalization RMSNorm                 # used before the blocks (both attn and mlp) throughout the model
--norm-epsilon 1e-5                     # "rms_norm_eps"
--disable-bias-linear                   # "mlp_bias": false, "attention_bias": false  -- following Llama-3.2-1B config
--transformer-impl transformer_engine
--seq-length 8192                       # runtime, doesn't affect the model's size but required to build a model

### MLP
--ffn-hidden-size 6208                  # shrunk from 8192 (97x64) to absorb the wider GDN-heavy mixer and match Llama-3.2-1B param count
--swiglu                                # "hidden_act": "silu"  -- following Llama-3.2-1B config + gating

### Token mixer -- hybrid: GDN on the linear-attention layers, gated softmax attn on the SDPA layers

# Linear-attention layers (12 of 16): GDN, identical to gdn_1B_args_8heads_ffn5824.txt
--experimental-attention-variant gated_delta_net
--linear-attention-freq ([1]*3+[0]*1)*4   # per-layer pattern: 1=GDN, 0=gated softmax attn -> [1,1,1,0]*4
                                          # = 12 GDN + 4 attn, attn at layers 3/7/11/15 (Qwen full_attention_interval=4)
                                          # equivalent shorthand: --linear-attention-freq 4
--linear-num-key-heads 8               # GDN's K/Q heads
--linear-num-value-heads 8             # GDN's V heads, must be a multiple of linear_num_key_heads
--linear-key-head-dim 192              # key_dim = 8*192 = 1536 = 0.75*hidden
--linear-value-head-dim 384            # value_dim = 8*384 = 3072 = 1.5*hidden = 2*key_dim
--linear-conv-kernel-dim 4

# SDPA layers (4 of 16): gated softmax attention, identical to gated_attn_llama_1B_args_ffn7488.txt
--num-attention-heads 32               # "num_attention_heads": 32 -- Llama-3.2-1B config
--group-query-attention
--num-query-groups 8                   # "num_key_value_heads": 8 -- Llama-3.2-1B config
--attention-output-gate               # per-head sigmoid gate on the attention output (extra Q-sized slot in linear_qkv); no effect on GDN layers
--position-embedding-type rope         # for the SDPA layers; GDN ignores positions (conv + decay gate), RoPE has no params so no count impact
--rotary-base 500000                   # "rope_theta": 500000.0 -- Llama-3.2-1B config
--rope-scaling-factor 1                 # pretraining from scratch | "factor": 32.0 in the Llama-3.2-1B config
--use-rope-scaling
--max-position-embeddings 8192          # 8192 since the scaling factor is 1 | 131072 in the Llama-3.2-1B config
# GDN internals (Q/K L2-norm, dt_bias, A_log, biases, gating) are hardcoded to the paper's values.
# Norms inside GDN follow the global --normalization RMSNorm / --norm-epsilon.
#
# Architecture check: count_model_param.slurm on this file confirms the hybrid -- the per-param dump
# has both gated-attn rows (linear_qkv/linear_proj) and GDN rows (in_proj/out_proj/conv1d/A_log/
# dt_bias/out_norm) under decoder.layers.*.self_attention.*, and the totals divide into 4 attn + 12
# GDN. Forward/no-crash is covered by the T3c 10-step smoke.
