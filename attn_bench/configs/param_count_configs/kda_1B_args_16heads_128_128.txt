############ KDA 1B CONFIG (128/128 head dims) #################
# Kimi Delta Attention: LLaMA-3.2-1B backbone with a KDA mixer.
# WHY 128/128, not the GDN-mirrored 192/384: fla's chunk_kda kernel produces NaN at 192/384
# (jobs 3244703 etc.); 128 head dim works (matches Kimi's real KDA layers + the shipped unit test).
# WHY 16 heads: 16 * 128 = 2048 = same total width as the softmax baseline's 32*64 attention block,
# so KDA param-matches Llama-1B with a normal FFN *shrink* (like GDN). 8 heads (the working smoke-test
# geometry, job 3245161) gives a 1024-wide mixer -- thinner than attention -- and would need FFN to grow.
# FFN left at 8192 (unshrunk) so the raw KDA param size is visible; run count_model_param, then shrink.
# Requires the PR #5769 cherry-pick + fla.ops.kda.chunk_kda.
# args names taken from: megatron/core/transformer/transformer_config.py

### Embeddings / input-output:
--make-vocab-size-divisible-by 128      # sizes the model's input / output embedding matrix (vocab x hidden / hidden x vocab)
# Llama-3.2-1B config: "vocab_size": 128256 -- derived from the tokenizer
# Llama-3.2-1B config: "tie_word_embeddings": true -- embeddings tied by default; need --untie-embeddings-and-output-weights to disable

### Global / model-wide
# not changed, same as Llama 1B config
--hidden-size 2048                      # representing 1 token
--num-layers 16                         # how many blocks
--normalization RMSNorm                 # used before the blocks (both attn and mlp) throughout the model
--norm-epsilon 1e-5                     # "rms_norm_eps"
--disable-bias-linear                   # "mlp_bias": false, "attention_bias": false  -- following Llama-3.2-1B config
--transformer-impl transformer_engine
--seq-length 8192                       # runtime, doesn't affect the model's size but required to build a model

### MLP
--ffn-hidden-size 8192                  # unshrunk (Llama default); shrink to param-match after running the counter
--swiglu                                # "hidden_act": "silu"  -- following Llama-3.2-1B config + gating

### Token mixer
--experimental-attention-variant kimi_delta_attention
--linear-attention-freq [1]*16
--linear-num-key-heads 16          # KDA's K/Q heads
--linear-num-value-heads 16        # KDA's V heads, must be a multiple of linear_num_key_heads
--linear-key-head-dim 128          # qk_dim = 16*128 = 2048 ; also the fine-grained gate_dim = key_head_dim * num_value_heads
--linear-value-head-dim 128        # v_dim = 16*128 = 2048 ; also the low-rank gate bottleneck dim (gate_low_rank_dim = value_head_dim)
--linear-conv-kernel-dim 4
--num-attention-heads 32           # required to build the model but is not used for KDA
# --group-query-attention / --num-query-groups -- not used in pure KDA
--position-embedding-type none     # KDA ignores positions (decay handles it); 'none' = no positional params at all
--max-position-embeddings 8192     # required by validation (must be >= seq-length), no effect
# KDA shares GDN's linear-attention validation. Gate low-rank dim, Q/K L2-norm, betas, biases and the
# gated output RMSNorm are hardcoded in kimi_delta_attention.py to the FLA reference values.
